Prime Map API 0.23 — Referans

Herhangi bir X-Plane 11 eklentisine gömülebilen harita kütüphanesi · 2026-11-12 · harita fabrikası, bağımsız çoklu harita, harita başına ayar dosyaları, ortak önbellek

1. Tanım

Prime Map API, X-Plane 11 eklentilerinin kendi arayüzlerine katmanlı bir hareketli harita yerleştirmesini sağlayan bir C++ kütüphanesidir (SHARED DLL: prime_map.dll). Harita, verisini sadece X-Plane'den alır: kurulu DSF sahneleri (zemin, su, yollar), apt.dat (havalimanları), nav veritabanı, earth_awy.dat (hava yolları), dataref'ler (uçak) ve FMS (rota). Her harita kendi XPLM penceresinde, kütüphanenin flight loop'uyla çizilir; host eklenti harita nesnesini fabrikadan alır, ayarlar, açar ve bütün davranışı API fonksiyonlarıyla yönetir.

ÖzellikDeğer
PlatformX-Plane 11, Windows x64, SDK 3.0 (XPLM301). Host her zaman bir X-Plane eklentisidir.
Dil, derleyiciC++20, MSVC 2022, /MD CRT (host da /MD olmak zorunda)
Ad alanı / başlıkprime / tek başlık PrimeMap.h (#include <prime_map_api/PrimeMap.h>; sadece standart C++ tipleri)
Dağıtımprime_map.dll host'un win.xpl dosyasının yanına kopyalanır. Hedef makinede VC_redist.x64 gerekir.
Harita sayısıSınırsız, hepsi bağımsız. Her haritanın kendi görünümü, modu, katmanları, çizimleri, uçak / rota kaynağı, zemin ayarları ve ayar dosyaları vardır.
Ortak olanVeri ve önbellek: nav, havalimanları, hava yolları, zemin disk önbelleği, bellekteki karolar, video belleğindeki dokular. Bir haritanın yüklediği karo diğerine de yarar. Sınırlar açık harita sayısıyla büyür (bölüm 9).
ThreadBütün fonksiyonlar X-Plane ana thread'inden çağrılır
Hata modeliİstisna yok. Geçersiz değer: işlem yapılmaz, Log'a tek satır WARNING, dönüş false / boş.

Adım 3 — my_host/CMakeLists.txt (tamamı)

cmake_minimum_required(VERSION 3.16)
cmake_policy(SET CMP0091 NEW)                      # MSVC runtime secimi
project(my_host VERSION 1.0 LANGUAGES CXX)

set(COTS_DIR "C:/workspace/cots" CACHE PATH "Ucuncu taraf bilesenler")
if(NOT EXISTS "${COTS_DIR}/X-Plane-SDK" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../cots/X-Plane-SDK")
    get_filename_component(COTS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../cots" ABSOLUTE)
    set(COTS_DIR "${COTS_DIR}" CACHE PATH "Ucuncu taraf bilesenler" FORCE)
endif()
set(XPSDK "${COTS_DIR}/X-Plane-SDK" CACHE PATH "X-Plane SDK")
set(PRIME_MAP_DIR "${CMAKE_SOURCE_DIR}/../prime_map_api" CACHE PATH "prime_map_api klasoru")
set(XPLANE_DIR "C:/X-Plane 11" CACHE PATH "X-Plane 11 klasoru")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")   # /MD: kutuphane ile AYNI olmali

add_subdirectory(${PRIME_MAP_DIR} prime_map_build)  # kutuphaneyi ayni oturumda derle

add_library(my_host SHARED src/MyHost.cpp)
target_include_directories(my_host SYSTEM PRIVATE ${XPSDK}/CHeaders/XPLM)
target_compile_definitions(my_host PRIVATE XPLM200=1 XPLM210=1 XPLM300=1 XPLM301=1 IBM=1 APL=0 LIN=0)
target_link_libraries(my_host PRIVATE prime_map ${XPSDK}/Libraries/Win/XPLM_64.lib opengl32)

# X-Plane eklenti kalibi: <eklenti>/64/win.xpl
set(OUT_DIR ${CMAKE_BINARY_DIR}/my-host)
set_target_properties(my_host PROPERTIES PREFIX "" OUTPUT_NAME win SUFFIX ".xpl"
    RUNTIME_OUTPUT_DIRECTORY ${OUT_DIR}/64 LIBRARY_OUTPUT_DIRECTORY ${OUT_DIR}/64)

# prime_map.dll her derlemede win.xpl'nin yanina (ve istenirse X-Plane'e)
add_custom_target(my_host_deploy ALL
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:prime_map> "${OUT_DIR}/64/"
    DEPENDS prime_map my_host)
option(MAP_DEPLOY "Derlemeden sonra X-Plane'e kopyala" ON)
if(MAP_DEPLOY AND EXISTS "${XPLANE_DIR}/Resources/plugins")
    set(DEPLOY_DIR "${XPLANE_DIR}/Resources/plugins/my-host")
    add_custom_command(TARGET my_host_deploy POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "${DEPLOY_DIR}/64"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:my_host> "${DEPLOY_DIR}/64/"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:prime_map> "${DEPLOY_DIR}/64/")
endif()

Adım 4 — my_host/src/MyHost.cpp (tamamı; iki haritalı çalışan eklenti)

#include <cstring>
#include <memory>

#include "XPLMDefs.h"
#include "XPLMMenus.h"
#include "XPLMPlugin.h"

#include <prime_map_api/PrimeMap.h>

namespace
{
std::unique_ptr<prime::Map> gMain;      // gomulu, host'un belirledigi yerde
std::unique_ptr<prime::Map> gOverview;  // ayri pencere, kullanici tasir
XPLMMenuID gMenu = nullptr;

void OnMenu(void* /*pMenuRef*/, void* pItemRef)
{
    const intptr_t tItem = reinterpret_cast<intptr_t>(pItemRef);
    if (tItem == 1 && gMain)     { gMain->SetVisible(!gMain->IsVisible()); }
    if (tItem == 2 && gOverview) { gOverview->SetVisible(!gOverview->IsVisible()); }
    if (tItem == 3 && gMain)     { gMain->NextMode(); }
    if (tItem == 4 && gMain)     { gMain->GoToAirport("KSEA"); }
}
}  // namespace

PLUGIN_API int XPluginStart(char* pOutName, char* pOutSignature, char* pOutDescription)
{
    std::strcpy(pOutName, "My Host");
    std::strcpy(pOutSignature, "example.my_host");
    std::strcpy(pOutDescription, "Prime Map API example host");
    return 1;
}

PLUGIN_API int XPluginEnable()
{
    // 1) Kutuphane (surecte tek; baska bir host daha once baslattiysa onun klasorleri gecerlidir)
    prime::MapLibrary& tLib = prime::MapLibrary::Get();
    prime::InitConfig tInit;
    tInit.mLogPrefix = "[my-host]";        // veri klasoru bos: .../Resources/plugins/my-host/
    if (!tLib.Init(tInit))
    {
        return 0;
    }

    // 2) Ana harita: fabrika -> ayarla -> ac
    gMain = tLib.CreateMap();
    gMain->SetName("main");                               // maps/main/
    gMain->SetStyle(prime::WindowStyle::Embedded);
    gMain->SetGeometry({100, 800, 900, 600});
    gMain->SetMode(prime::MapMode::Scenery);
    gMain->SetCenter(47.45, -122.30);
    gMain->SetZoom(10.0);
    gMain->SetMouseInput(true);
    gMain->Open();

    // 3) Ikinci harita: bagimsiz
    gOverview = tLib.CreateMap();
    gOverview->SetName("overview");                       // maps/overview/
    gOverview->SetStyle(prime::WindowStyle::Standalone);
    gOverview->SetTitle("Overview");
    gOverview->SetMode(prime::MapMode::IfrHigh);
    gOverview->SetUiVisible(true);
    gOverview->SetMouseInput(true);
    gOverview->Open();

    // 4) Host'un kendi arayuzu (burada: menu)
    const int tSlot = XPLMAppendMenuItem(XPLMFindPluginsMenu(), "My Host", nullptr, 0);
    gMenu = XPLMCreateMenu("My Host", XPLMFindPluginsMenu(), tSlot, OnMenu, nullptr);
    XPLMAppendMenuItem(gMenu, "Show / hide main map", reinterpret_cast<void*>(intptr_t(1)), 0);
    XPLMAppendMenuItem(gMenu, "Show / hide overview", reinterpret_cast<void*>(intptr_t(2)), 0);
    XPLMAppendMenuItem(gMenu, "Next mode (main)", reinterpret_cast<void*>(intptr_t(3)), 0);
    XPLMAppendMenuItem(gMenu, "Go to KSEA (main)", reinterpret_cast<void*>(intptr_t(4)), 0);
    return 1;
}

PLUGIN_API void XPluginDisable()
{
    if (gMenu) { XPLMDestroyMenu(gMenu); gMenu = nullptr; }
    gOverview.reset();                    // once haritalar (ayarlari yazilir)
    gMain.reset();
    prime::MapLibrary::Get().Shutdown();  // sonra kutuphane
}

PLUGIN_API void XPluginStop() {}
PLUGIN_API void XPluginReceiveMessage(XPLMPluginID, int, void*) {}

3.2 Dosyalar ve klasörler

Hepsi veri klasörünün altındadır. Veri klasörü: InitConfig::mDataDirectory; boşsa host eklentisinin klasörü (<X-Plane>/Resources/plugins/<host>/).

<veri klasoru>/
  prime_map.ini                    merkezi ayarlar (butun haritalar; bolum 9)
  maps/
    ios_left/                      SetName("ios_left") verilen harita
      map_settings.ini             pencere, gorunum, mod, katmanlar, zoom sinirlari, kaynaklar (bolum 8)
      map_drawings.txt             kullanici cizimleri
      terrain_colors.ini           bu haritanin zemin renkleri / ayarlari
    map_2/                         ad verilmeyen harita: "map_<kimlik>"
      ...
  cache/                           ORTAK onbellek (silinebilir)
    terrain/                       *.terrain *.elevation *.vectors *.nobase, xplane_palette.txt
    airports/                      airport_index.bin
KonuKural
Harita adıSetName ile verilir. Klasör adında harf, rakam, - _ . dışındaki karakterler _ olur. Ad verilmezse klasör map_<kimlik> (kimlik = oluşturulma sırası, 1'den başlar; haritalar hep aynı sırayla oluşturuluyorsa klasör de aynı kalır). Kalıcı ayar isteyen host ad vermelidir.
Aynı adİki canlı harita aynı klasörü kullanamaz: ikinci SetName false döner.
Ne zaman okunurOpen() sırasında (sadece Persistence::LoadAndSave).
Ne zaman yazılırİlk Open()'da (dosyalar oluşsun); bir ayar değişince ([maps] autosave, en sık autosave_seconds arayla); Close(), harita bırakılırken, Shutdown; Map::SaveSettings().
ÖnbellekTek yerde, bütün haritalar için ortak. InitConfig::mCacheDirectory ile başka yere alınabilir.
Birden çok hostAynı X-Plane sürecinde kütüphane tektir: ilk Init eden host'un klasörleri geçerlidir (diğerlerinin InitConfig'i yok sayılır, Log'a yazılır). Hostların haritaları aynı maps/ altında durur: adları farklı seçin.

Görünürlük kuralları

  • Üst öğeyi kapatmak alt öğeleri gizler; alt öğelerin kendi durumu korunur.
  • Kapalı katmanın bir alt katmanını açmak sadece o alt katmanı açar (katman Mixed).
  • Son alt katman kapanınca katman da kapanır; bütün alt katmanları kapalı bir katman açılınca hepsi açılır.
  • Grup açmak sadece o modda varsayılan açık katmanları açar. Durum mod başına ve harita başına saklanır.

Zoom aralığı ("şu uzaklıktan itibaren görünsün")

Her katmanın yerleşik eşikleri vardır (örnek: pistler zoom 11+, VOR 5+, NDB 7+, fix 9+, hava yolları 6.5+, yollar 7.5+). SetLayerZoomRange bunları daraltır (erkene alamaz); kutu durumu (IsLayerVisible) değişmez.

map->SetLayerZoomRange(prime::MapLayer::AirwaysGroup, 8.0, 99.0);   // hava yollari ancak zoom 8'den yakinda
map->SetLayerZoomRange(prime::MapLayer::Fixes, 10.5, 99.0);         // fix'ler 10.5'ten itibaren
map->SetLayerZoomRange(prime::MapLayer::Terrain, 0.0, 12.0);        // zemin 12'den yakinda cizilmez
map->SetLayerZoomRange(prime::MapLayer::RoadsLocal, 13.0, 99.0);    // alt katman: yerel yollar 13+
map->ClearLayerZoomRange(prime::MapLayer::Fixes);                   // siniri kaldir

s.mTerrainCacheMb = 4096;              // harita basina taban 4 GB
s.mCacheGrowthPercentPerMap = 10;
prime::MapLibrary::Get().SetSettings(s);   // hemen uygulanir + prime_map.ini'ye yazilir
# prime_map.ini (tam ornek)
[cache]
growth_percent_per_map=25
max_terrain_mb=0
prefetch_tiles=20
terrain_mb=1024

[maps]
autosave=1
autosave_seconds=2

[memory]
growth_percent_per_map=50
max_texture_mb=2048
texture_mb=512

Geçerli aralıklar (dışındaki değer en yakın sınıra çekilir): terrain_mb ≥ 0; growth_percent_per_map 0–1000; prefetch_tiles 0–500; texture_mb 64–16384; autosave_seconds 0–3600. Tavan tabandan küçükse taban geçerli olur.

Dosya ile InitConfig çakışırsa dosya kazanır (kullanıcının ayarı); InitConfig sadece dosyada olmayan anahtarların ilk değerini verir.

11. Tarifler

11.1 Gömülü harita: host penceresiyle birlikte yaşamak

Embedded harita ayrı bir XPLM penceresidir; host'un penceresine "yapışık" görünmesi host'un işidir:

KonuYapılacak
Yer ve boyutHost her karede (kendi çizim callback'inde ya da flight loop'ta) kendi penceresinin yerini XPLMGetWindowGeometry ile okur, haritaya ayırdığı dikdörtgeni hesaplar ve değiştiyse SetGeometry çağırır.
Göster / gizleHost penceresi gizlenince SetVisible(false); gösterilince SetVisible(true).
Üstte kalmaHost penceresi öne alındığında (tıklama) BringToFront(): harita host penceresinin üstünde kalır.
FareSetMouseInput(false): haritaya yapılan tıklama / tekerlek alttaki pencereye (host'a) geçer; host kendi düğmeleriyle ZoomIn, PanLeft ... çağırır. true: sürükle / tekerlek / tooltip haritada.
Yerleşik arayüzSetUiVisible(false): sadece harita. true: üst şerit, katman paneli, arama, durum şeridi de gelir.
Ekran ↔ coğrafyaHost kendi tıklamasını haritaya çevirmek için ScreenToGeo(x, y), bir noktayı işaretlemek için GeoToScreen kullanır (global boxel).
Pop-outEmbedded harita pop-out olmaz. Host penceresi ayrı işletim sistemi penceresine alınıyorsa harita için Standalone stili kullanın.
// host'un cizim callback'i ya da flight loop'u
int l, t, r, b;
XPLMGetWindowGeometry(hostWindow, &l, &t, &r, &b);
const prime::Geometry wanted{l + 10, t - 40, (r - l) - 20, (t - b) - 100};   // host icindeki harita alani
const prime::Geometry now = map->GetGeometry();
if (now.mLeft != wanted.mLeft || now.mTop != wanted.mTop || now.mWidth != wanted.mWidth || now.mHeight != wanted.mHeight)
{
    map->SetGeometry(wanted);
}

Çalışan örnek: plugin-api-test/src/MapUiTestMain.cpp.

Host penceresine gömülü harita, host düğmeleriyle

map->SetStyle(prime::WindowStyle::Embedded);
map->SetUiVisible(false);  map->SetMouseInput(false);    // butun kontrol host'ta
map->Open();
// host'un dugmeleri:           map->ZoomIn();  map->PanLeft();  map->NextMode();
// host penceresi tasininca:    map->SetGeometry({x, y, w, h});

Aynı uçuşun iki görünümü

overview->SetMode(prime::MapMode::IfrHigh);  overview->SetZoom(6);   overview->SetFollowAircraft(true);
detail->SetMode(prime::MapMode::Scenery);    detail->SetZoom(14);    detail->SetFollowAircraft(true);
detail->ShowOnlyLayers({prime::MapLayer::Terrain, prime::MapLayer::Runways, prime::MapLayer::Surfaces,
                        prime::MapLayer::Aircraft});

Haritaya kimlikle erişim

const int id = map->GetId();
if (prime::Map* m = prime::MapLibrary::Get().FindMap(id))        m->SetMode(prime::MapMode::Dark);
if (prime::Map* m = prime::MapLibrary::Get().FindMap("overview")) m->SetVisible(false);
for (prime::Map* m : prime::MapLibrary::Get().GetMaps())          m->SetMaxRenderFps(10);

Bir haritanın ayarını diğerine kopyalama

13. Sınırlar

  • Zemin işçisi tektir: birbirinden uzak yerlere bakan çok sayıda harita aynı işçiyi paylaşır (istekler her haritanın kendi merkezine göre sıralanır).
  • Bölge ön önbelleği (Precache...) aynı anda tek alan içindir (son çağrı geçerli).
  • 32'den fazla haritada doku işaretleri paylaşılır (sonuç yalnızca gereksiz yeniden istek).
  • SetXPlaneFont(true) iken harita resmi önbelleği kullanılmaz.
  • 180. boylam civarı kısmen ele alınır; yazılar ASCII.
